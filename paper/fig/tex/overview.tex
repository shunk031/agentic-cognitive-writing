  io/.style={box, minimum width=0.82\columnwidth},
  input/.style={io, fill=RoleNeutralFill, draw=RoleNeutralStroke},
  output/.style={io, fill=RoleNeutralFill, draw=RoleNeutralStroke, double, double distance=0.6pt},
  agent/.style={box, rounded corners=2pt, minimum width=0.25\columnwidth},
  mainagent/.style={agent, fill=RoleMonitorFill, draw=RoleMonitorStroke, line width=0.7pt},
  planner/.style={agent, fill=RolePlannerFill, draw=RolePlannerStroke},
  translator/.style={agent, fill=RoleTranslatorFill, draw=RoleTranslatorStroke},
  reviewer/.style={agent, fill=RoleReviewerFill, draw=RoleReviewerStroke},
  state/.style={box, rounded corners=1.5pt, dashed, fill=RoleStateFill, draw=RoleStateStroke, minimum width=0.82\columnwidth},

\node[input] (task) {\textbf{Input: writing task}\\instruction $I$ \; + \; context $C$ \; + \; output requirements $O$};
\node[mainagent, below=of task] (monitor) {\textbf{Monitor}\\select next process};

\node[translator, below=7mm of monitor] (translator) {\textbf{Translator}};
\node[planner, left=2.5mm of translator] (planner) {\textbf{Planner}};
\node[reviewer, right=2.5mm of translator] (reviewer) {\textbf{Reviewer}};

\node[state, below=7mm of translator] (state) {\textbf{Persistent writing state}\\draft $D_t$ \; | \; goals $G_t$ \; | \; history $H_t$};
\node[output, below=of state] (output) {\textbf{Output: completed document} $D_{\mathrm{final}}$};

\draw[flow] (task) -- (monitor);
\draw[flow] (monitor) -- (planner);
\draw[flow] (monitor) -- (translator);
\draw[flow] (monitor) -- (reviewer);
\draw[flow] (planner) -- (state);
\draw[flow] (translator) -- (state);
\draw[flow] (reviewer) -- (state);
\draw[loop] (state.west) -- ++(-7mm,0) |- (monitor.west);
\draw[flow, dashed] (state) -- node[right, font=\scriptsize]{terminate after review} (output);

\end{tikzpicture}
\caption{Overview of \condAgenticCogWriter. Inspired by the cognitive process theory of writing, a coding-agent-style control loop turns the task into a document via Planning, Translating, or Reviewing, selected online from the evolving document state.}
