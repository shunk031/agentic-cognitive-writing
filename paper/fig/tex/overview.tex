\begin{figure}[t]
\centering
\begin{tikzpicture}[
  >=Latex,
  node distance=4.5mm and 3mm,
  box/.style={draw, rounded corners=1.5pt, align=center, inner sep=3.2pt, font=\small},
  agent/.style={box, minimum width=0.25\columnwidth},
  state/.style={box, minimum width=0.82\columnwidth},
  io/.style={box, minimum width=0.82\columnwidth},
  flow/.style={->, line width=0.55pt},
  loop/.style={->, line width=0.55pt, rounded corners=4pt}
]

\node[io] (task) {\textbf{Writing task}\\instruction $I$ \; + \; context $C$ \; + \; output contract $O$};
\node[agent, below=of task] (monitor) {\textbf{Monitor}\\select next process};

\node[agent, below=7mm of monitor] (translator) {\textbf{Translator}\\\texttt{Translating}};
\node[agent, left=2.5mm of translator] (planner) {\textbf{Planner}\\\texttt{Planning}};
\node[agent, right=2.5mm of translator] (reviewer) {\textbf{Reviewer}\\\texttt{Reviewing}};

\node[state, below=7mm of translator] (state) {\textbf{Persistent writing state}\\draft $D_t$ \; | \; goals $G_t$ \; | \; history $H_t$};
\node[io, below=of state] (output) {\textbf{Completed document} $D_T$};

\draw[flow] (task) -- (monitor);
\draw[flow] (monitor) -- (planner);
\draw[flow] (monitor) -- (translator);
\draw[flow] (monitor) -- (reviewer);
\draw[flow] (planner) -- (state);
\draw[flow] (translator) -- (state);
\draw[flow] (reviewer) -- (state);
\draw[loop] (state.west) -- ++(-7mm,0) |- (monitor.west);
\draw[flow, dashed] (state) -- node[right, font=\scriptsize]{terminate after review} (output);

\end{tikzpicture}
\caption{Overview of \condAgenticCogWriter. The \texttt{Monitor} reads the persistent writing state, selects one writing process, delegates it to the corresponding specialist, and then observes the updated state. When the \texttt{Monitor} terminates after review, the current draft is returned as the completed document.}
\label{fig:agentic-cogwriter}
\end{figure}
