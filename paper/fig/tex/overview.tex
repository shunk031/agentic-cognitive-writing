\begin{tikzpicture}[
  >=Latex,
  node distance=4.5mm and 3mm,
  box/.style={draw, align=center, inner sep=3.2pt, font=\small},
  io/.style={box, minimum width=0.82\columnwidth},
  input/.style={io, fill=gray!12},
  output/.style={io, fill=green!10, double, double distance=0.6pt},
  agent/.style={box, rounded corners=2pt, minimum width=0.25\columnwidth},
  mainagent/.style={agent, fill=blue!16, line width=0.65pt},
  subagent/.style={agent, fill=blue!7},
  state/.style={box, rounded corners=1.5pt, dashed, fill=orange!10, minimum width=0.82\columnwidth},
  flow/.style={->, line width=0.55pt},
  loop/.style={->, line width=0.55pt, rounded corners=4pt}
]

\node[input] (task) {\textbf{Input: writing task}\\instruction $I$ \; + \; context $C$ \; + \; output requirements $O$};
\node[mainagent, below=of task] (monitor) {\textbf{Monitor}\\select next process};

\node[subagent, below=7mm of monitor] (translator) {\textbf{Translator}\\\texttt{Translating}};
\node[subagent, left=2.5mm of translator] (planner) {\textbf{Planner}\\\texttt{Planning}};
\node[subagent, right=2.5mm of translator] (reviewer) {\textbf{Reviewer}\\\texttt{Reviewing}};

\node[state, below=7mm of translator] (state) {\textbf{Persistent writing state}\\draft $D_t$ \; | \; goals $G_t$ \; | \; history $H_t$};
\node[output, below=of state] (output) {\textbf{Output: completed document} $D_{\mathrm{final}}$};

\draw[flow] (task) -- (monitor);
\draw[flow] (monitor) -- (planner);
\draw[flow] (monitor) -- (translator);
\draw[flow] (monitor) -- (reviewer);
\draw[flow] (planner) -- (state);
\draw[flow] (translator) -- (state);
\draw[flow] (reviewer) -- (state);
\draw[loop] (state.west) -- ++(-7mm,0) |- (monitor.west);
\draw[flow, dashed] (state) -- node[right, font=\scriptsize]{terminate after review} (output);

\end{tikzpicture}
\caption{Overview of \condAgenticCogWriter. Inspired by the cognitive process theory of writing~\citep{flower1981cognitive}, a coding-agent-style control loop turns the task into a document, with \texttt{Planning}, \texttt{Translating}, or \texttt{Reviewing} selected online from the evolving document state.}
\label{fig:agentic-cogwriter}
\end{figure}
